### 🛡️ ЭТАП 4: УСИЛЕНИЕ БЕЗОПАСНОСТИ (12-24 часа)

#### 🔒 Жесткая конфигурация безопасности
```nginx
# /etc/nginx/sites-available/vk.com
server {
    listen 443 ssl http2;
    server_name vk.com;
    
    # Security Headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    add_header Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'; img-src 'self' data: https:; font-src 'self'; connect-src 'self'; frame-ancestors 'none'; base-uri 'self'; form-action 'self'" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Permissions-Policy "geolocation=(), microphone=(), camera=()" always;
    
    # WAF Rules
    location ~* \.(php|jsp|asp|aspx)$ {
        if ($args ~* "union.*select|or.*1.*=.*1|script|\.\/") {
            return 403;
        }
        if ($request_method !~ ^(GET|POST|HEAD)$) {
            return 405;
        }
        if ($http_user_agent ~* (sqlmap|nmap|nikto|burp)) {
            return 403;
        }
    }
    
    # Rate Limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
    limit_req_zone $binary_remote_addr zone=login:10m rate=1r/s;
    
    location /api/login {
        limit_req zone=login burst=5 nodelay;
        try_files $uri $uri/ /index.php?$query_string;
    }
    
    location /api/ {
        limit_req zone=api burst=20 nodelay;
        try_files $uri $uri/ /index.php?$query_string;
    }
}
```

### 📊 Система мониторинга атак
```bash
# Установка fail2ban
sudo apt install fail2ban -y

# Настройка правил для nginx
cat > /etc/fail2ban/jail.local << EOF
[nginx-http-auth]
enabled = true
filter = nginx-http-auth
logpath = /var/log/nginx/error.log
maxretry = 3
bantime = 3600

[nginx-limit-req]
enabled = true
filter = nginx-limit-req
logpath = /var/log/nginx/error.log
maxretry = 10
bantime = 600

[nginx-noscript]
enabled = true
filter = nginx-noscript
logpath = /var/log/nginx/access.log
maxretry = 6
bantime = 86400
EOF

sudo systemctl enable fail2ban
sudo systemctl start fail2ban
```

### 📊 Real-time мониторинг
```bash
# Скрипт мониторинга атак
cat > /usr/local/bin/security_monitor.sh << 'EOF'
#!/bin/bash

LOG_FILE="/var/log/nginx/access.log"
ALERT_THRESHOLD=10
ATTACK_WINDOW=60

while true; do
    # Подсчет атак за последнюю минуту
    ATTACK_COUNT=$(tail -n 1000 "$LOG_FILE" | grep -E "(union.*select|script|\.\/)" | grep "$(date '+%d/%b/%Y:%H:%M')" | wc -l)
    
    if [ "$ATTACK_COUNT" -gt "$ALERT_THRESHOLD" ]; then
        echo "ALERT: $ATTACK_COUNT attacks detected in last minute!"
        # Отправка уведомления
        curl -X POST "https://alerts.company.com/webhook" \
            -H "Content-Type: application/json" \
            -d "{\"message\": \"High attack volume: $ATTACK_COUNT attacks/min\", \"severity\": \"critical\"}"
    fi
    
    sleep 30
done
EOF

chmod +x /usr/local/bin/security_monitor.sh
systemctl enable security_monitor
systemctl start security_monitor
```
