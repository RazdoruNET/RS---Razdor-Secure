## ⚡ НЕЗАМЕДЛИТЕЛЬНЫЕ ДЕЙСТВИЯ (0-5 минут)

### 🛡️ ШАГ 1: ОТКЛЮЧЕНИЕ ВСЕХ СЕРВИСОВ

```bash
# НЕМЕДЛЕННОЕ отключение веб-серверов
sudo systemctl stop nginx
sudo systemctl stop apache2
sudo systemctl stop lighttpd

# Отключение PHP
sudo systemctl stop php-fpm
sudo systemctl stop php7.4-fpm
sudo systemctl stop php8.0-fpm

# Отключение базы данных
sudo systemctl stop mysql
sudo systemctl stop postgresql
sudo systemctl stop redis-server

# Отключение внешних соединений
sudo iptables -A INPUT -j DROP
sudo iptables -I INPUT -s 127.0.0.1 -j ACCEPT
```

### 🔒 ШАГ 2: ИЗОЛЯЦИЯ СЕТИ

```bash
# Блокировка всех внешних соединений
sudo ufw deny incoming
sudo ufw deny outgoing

# Отключение сетевых интерфейсов (кроме management)
sudo ifdown eth0
sudo ifdown eth1

# Включение только management сети
sudo ifup eth2
```

## 📋 ШАГ 3: СОХРАНЕНИЕ КРИТИЧЕСКИХ ДАННЫХ

### 💾 Резервное копирование баз данных
```bash
# Создание директории для бэкапов
mkdir -p /mnt/secure_backup/$(date +%Y%m%d_%H%M%S)
cd /mnt/secure_backup/$(date +%Y%m%d_%H%M%S)

# Дамп баз данных (если еще работает)
mysqldump --single-transaction --routines --triggers --all-databases > full_database_backup.sql
pg_dumpall > full_postgresql_backup.sql

# Копирование файлов баз данных
sudo cp -r /var/lib/mysql ./mysql_files/
sudo cp -r /var/lib/postgresql ./postgresql_files/

# Архивирование
tar -czf database_backup_$(date +%Y%m%d_%H%M%S).tar.gz *.sql *_files/
```

### 📁 Резервное копирование файлов приложений
```bash
# Копирование веб-файлов
sudo cp -r /var/www/html ./web_files/
sudo cp -r /opt/applications ./app_files/

# Копирование конфигураций
sudo cp -r /etc/nginx ./nginx_configs/
sudo cp -r /etc/apache2 ./apache_configs/
sudo cp -r /etc/php ./php_configs/

# Копирование логов
sudo cp -r /var/log ./system_logs/

# Архивирование
tar -czf application_backup_$(date +%Y%m%d_%H%M%S).tar.gz *_files/ *_configs/ system_logs/
```

## 🔍 ШАГ 4: FORENSIC СБОР ДАННЫХ

### 📊 Сбор сетевой информации
```bash
# Создание forensic директории
mkdir -p /mnt/forensic/$(date +%Y%m%d_%H%M%S)
cd /mnt/forensic/$(date +%Y%m%d_%H%M%S)

# Сохранение сетевых соединений
netstat -tulnp > active_connections.txt
ss -tulnp > active_connections_ss.txt
lsof -i > open_ports.txt

# Сбор ARP таблицы
arp -a > arp_table.txt
ip neigh show > neighbor_table.txt

# Сбор маршрутизации
route -n > routing_table.txt
ip route show > routing_table_ip.txt
```

### 🔍 Сбор информации о процессах
```bash
# Список процессов
ps aux > running_processes.txt
ps -eo pid,ppid,cmd,%mem,%cpu --sort=-%mem > processes_by_memory.txt
ps -eo pid,ppid,cmd,%mem,%cpu --sort=-%cpu > processes_by_cpu.txt

# Список открытых файлов
lsof > open_files.txt
lsof -i > network_files.txt

# Проверка на руткиты
sudo rkhunter --checkall --sk --report-file=rkhunter_report.txt
sudo chkrootkit > chkrootkit_report.txt
```

### 📋 Сбор логов атак
```bash
# Анализ веб-логов
grep -i "union.*select\|or.*1.*=.*1" /var/log/nginx/access.log > sql_injection_attacks.log
grep -i "<script\|onerror\|onload" /var/log/nginx/access.log > xss_attacks.log
grep -i "\.\./\.\./\.\." /var/log/nginx/access.log > path_traversal_attacks.log

# Анализ системных логов
grep -i "failed\|error\|attack" /var/log/auth.log > auth_failures.log
grep -i "intrusion\|breach\|compromise" /var/log/syslog > system_alerts.log

# Анализ логов приложений
find /var/log -name "*.log" -exec grep -l "error\|exception\|attack" {} \; > suspicious_logs.txt
```

## 🚨 ШАГ 5: ИДЕНТИФИКАЦИЯ КОМПРОМЕТАЦИИ

### 🔍 Проверка целостности файлов
```bash
# Проверка критических файлов
md5sum /etc/passwd > passwd_md5.txt
md5sum /etc/shadow > shadow_md5.txt
md5sum /var/www/html/config/database.php > config_md5.txt

# Поиск измененных файлов
find /var/www/html -type f -mtime -1 -ls > recently_modified_files.txt
find /etc -type f -mtime -1 -ls > recently_modified_configs.txt

# Поиск подозрительных файлов
find /var/www/html -name "*.php" -size +10k -ls > large_php_files.txt
find /tmp -name "*.php" -ls > tmp_php_files.txt
find /dev/shm -name "*.php" -ls > shm_php_files.txt
```

### 🕵️ Поиск бэкдоров и веб-шеллов
```bash
# Поиск веб-шеллов
grep -r "eval.*\$_POST\|system.*\$_GET\|shell_exec" /var/www/html/ > potential_shells.txt
grep -r "base64_decode\|gzuncompress\|str_rot13" /var/www/html/ > encoded_payloads.txt

# Поиск бэкдоров
find /var/www/html -name "*.php" -exec grep -l "backdoor\|shell\|cmd" {} \; > backdoor_files.txt
find /var/www/html -name "*.php" -exec grep -l "eval.*base64" {} \; > base64_shells.txt

# Проверка cron заданий
crontab -l > current_cron.txt
cat /etc/crontab > system_cron.txt
ls -la /etc/cron.* > cron_directories.txt
```

## 🛡️ ШАГ 7: ВОССТАНОВЛЕНИЕ БЕЗОПАСНОСТИ

### 🔒 Изменение паролей
```bash
# Изменение паролей системных пользователей
sudo passwd root
sudo passwd $(whoami)

# Изменение паролей баз данных
mysql -u root -p -e "SET PASSWORD FOR 'root'@'localhost' = PASSWORD('$(openssl rand -base64 32)');"
mysql -u root -p -e "SET PASSWORD FOR 'app_user'@'localhost' = PASSWORD('$(openssl rand -base64 32)');"

# Изменение SSH ключей
ssh-keygen -t rsa -b 4096 -f ~/.ssh/emergency_key
chmod 600 ~/.ssh/emergency_key
```

### 🛠️ Удаление вредоносного кода
```bash
# Удаление найденных бэкдоров
while read file; do
    echo "Удаление вредоносного файла: $file"
    sudo rm -f "$file"
done < backdoor_files.txt

# Очистка временных директорий
sudo rm -rf /tmp/*.php
sudo rm -rf /dev/shm/*.php
sudo rm -rf /var/tmp/*.php

# Блокировка подозрительных IP
while read ip; do
    echo "Блокировка IP: $ip"
    sudo iptables -A INPUT -s "$ip" -j DROP
done < suspicious_ips.txt
```
